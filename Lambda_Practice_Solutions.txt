n % 2 == 0 else “Positive Odd” if n > 0 else (“Negative Even” if n % 2
== 0 else “Negative Odd”))

print(classify(8)) print(classify(-3)) print(classify(0))

2. Second-Largest of Three Numbers

second_largest = lambda a, b, c: a if (a > b and a < c) or (a < b and
a > c) else (b if (b > a and b < c) or (b < a and b > c) else c)

print(second_largest(10, 25, 15))

11. Armstrong Numbers

def is_armstrong(n): digits = str(n) power = len(digits) return n >= 0
and sum(int(d) ** power for d in digits) == n

numbers = [0, 1, 153, 370, 123, 9474] result = list(filter(lambda n:
is_armstrong(n), numbers)) print(result)

25. Filter -> Map -> Sort

def is_prime(n): return n > 1 and all(n % i != 0 for i in range(2, int(n
** 0.5) + 1))

numbers = [2, 3, 4, 5, 7, 11, 13, 17, 19, 23] prime_numbers =
filter(lambda n: is_prime(n), numbers) transformed = map(lambda n:
sum(int(d) ** 2 for d in str(n)), prime_numbers) result =
sorted(set(transformed), reverse=True) print(result)

29. Advanced Transaction Analysis

transactions = [ {“transaction_id”: “T1”, “amount”: 1200, “status”:

“SUCCESS”}, {“transaction_id”: “T3”, “amount”: 800, “status”: “FAILED”},
{“transaction_id”: “T4”, “amount”: 750, “status”: “SUCCESS”}] valid =
filter(lambda t: t[“status”] == “SUCCESS” and t[“amount”] >= 500,
transactions) ordered = sorted(valid, key=lambda t: t[“amount”],
reverse=True) total = reduce(lambda acc, t: acc + t[“amount”], ordered,
0) print(ordered) print(“Total transaction value:”, total)

30. Ultimate Functional Programming Challenge

def is_prime(n): return n > 1 and all(n % i != 0 for i in range(2, int(n
** 0.5) + 1))

digit_sum = lambda n: sum(int(d) for d in str(n)) digit_product = lambda
n: reduce(lambda a, b: a * b, (int(d) for d in str(n)), 1)

numbers = [12, 15, 23, 30, 41, 56, 78, 101, -9, 0, 123] positive =
filter(lambda n: n > 0, numbers) valid = filter(lambda n:
is_prime(digit_sum(n)), positive) products = map(lambda n:
digit_product(n), valid) unique_products = set(products) result =
sorted(unique_products, key=lambda n: (digit_sum(n), -n)) total =
reduce(lambda a, b: a + b, result, 0) print(“Sorted list:”, result)
print(“Final sum:”, total)
