
EXAMPLE 1: ZeroDivisionError

try:
    a = 20
    b = 0
    print(a / b)
except ZeroDivisionError:
    print("Cannot divide by zero")


EXAMPLE 2: ValueError

try:
    num = int(input("Enter a number: "))
    print("Number:", num)
except ValueError:
    print("Please enter a valid integer")


EXAMPLE 3: IndexError

try:
    numbers = [10, 20, 30, 40, 50]
    print(numbers[8])
except IndexError:
    print("Invalid list index")


EXAMPLE 4: KeyError

try:
    student = {"name": "Rahul", "age": 21}
    print(student["marks"])
except KeyError:
    print("Key does not exist")


EXAMPLE 5: TypeError

try:
    a = 100
    b = "Python"
    print(a + b)
except TypeError:
    print("Invalid operation between different data types")


EXAMPLE 6: FileNotFoundError

try:
    file = open("details.txt", "r")
    print(file.read())
    file.close()
except FileNotFoundError:
    print("File not found")


EXAMPLE 7: Multiple except Blocks

try:
    a = int(input("Enter first number: "))
    b = int(input("Enter second number: "))
    print("Result:", a / b)
except ValueError:
    print("Enter numbers only")
except ZeroDivisionError:
    print("Cannot divide by zero")


EXAMPLE 8: try, except and else

try:
    a = 50
    b = 10
    result = a / b
except ZeroDivisionError:
    print("Cannot divide by zero")
else:
    print("Division successful")
    print("Result:", result)


EXAMPLE 9: finally Block

try:
    print("Program started")
    result = 15 / 0
except ZeroDivisionError:
    print("Division by zero error")
finally:
    print("Program completed")


EXAMPLE 10: raise Keyword

try:
    marks = 25

    if marks < 35:
        raise ValueError("Student has failed")

    print("Student has passed")
except ValueError as e:
    print("Error:", e)
