import pandas as pd

                "Status":["Manager","Designer","Intern","Intern"],
                "birthyear":[1978,1988,2002,2004]})

print(df)
res=tuple(filter(lambda x:x > 40,df['age']))

import pandas as pd


df = pd.read_csv("patient_records.csv")


print(df)


print("\nAge above 50:")
print(df[df["Age"] > 45])


print("\nMale:")
print(df[df["Sex"].apply(lambda x: x == "M")])


print("\nFemale:")
print(df[df["Sex"].apply(lambda x: x == "F")])

print("\nHeart disease:")
print(df[df["Diagnosis"].apply(lambda x: x == "Positive")])


df = df.fillna("None")
print("\nMissing values replaced:")
print(df)

df["VisitDate"] = pd.to_datetime(df["VisitDate"])
print("\nDates after January 1, 2026:")
print(df[df["VisitDate"].apply(lambda x: x > pd.Timestamp("2026-01-01"))])
