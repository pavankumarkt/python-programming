12. Check whether a number is an Armstrong number

num = int(input("Enter a number: "))
original = num
temp = num
count = 0

while temp > 0:
    count = count + 1
    temp = temp // 10

temp = num
total = 0

while temp > 0:
    digit = temp % 10
    total = total + digit ** count
    temp = temp // 10

if total == original:
    print("Armstrong number")
else:
    print("Not an Armstrong number")

20. Demonstrate mutable and immutable data types

# List is mutable
my_list = [10, 20, 30]
my_list[0] = 100
print("List:", my_list)

# Tuple is immutable
my_tuple = (10, 20, 30)
print("Tuple:", my_tuple)

# String is immutable
my_string = "Hello"
print("String:", my_string)

# Set is mutable
my_set = {10, 20, 30}
my_set.add(40)
print("Set:", my_set)

# List and set can be changed after creation.
# Tuple and string cannot be changed directly.
